USE nishijuku;

DROP TABLE IF EXISTS event_images;
CREATE TABLE event_images (

id INT(11) AUTO_INCREMENT NOT NULL COMMENT '画像ID', PRIMARY KEY (id),
event_id INT(11) NOT NULL COMMENT 'イベントID',
description TEXT COMMENT '説明',
image_path VARCHAR(255) NOT NULL COMMENT '写真パス',
display_order INT(2) NOT NULL COMMENT '表示順',
create_by VARCHAR(100) COMMENT '投稿者',
create_at DATETIME COMMENT '投稿時間'
)ENGINE=InnoDB ROW_FORMAT=DYNAMIC COLLATE=utf8mb4_general_ci COMMENT='イベント写真';
