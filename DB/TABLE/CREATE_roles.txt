USE nishijuku;

-- 既にあれば削除
DROP TABLE IF EXISTS roles;

-- roles テーブル作成
CREATE TABLE roles (
  id INT(11) NOT NULL COMMENT '権限ID',
  display_order INT(5) NOT NULL COMMENT '表示順',
  role_name VARCHAR(20) NOT NULL COMMENT '権限名',
  description VARCHAR(50) COMMENT '権限説明',
  PRIMARY KEY (id)
)
ENGINE=InnoDB
ROW_FORMAT=DYNAMIC
COLLATE=utf8mb4_general_ci
COMMENT='権限';
