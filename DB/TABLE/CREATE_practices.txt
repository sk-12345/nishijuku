USE nishijuku;

DROP TABLE IF EXISTS practices;
CREATE TABLE practices (

id INT(11) AUTO_INCREMENT NOT NULL COMMENT 'ID', PRIMARY KEY (id),
title VARCHAR(100) NOT NULL COMMENT 'タイトル', INDEX (title),
description TEXT NOT NULL COMMENT '説明',
create_by INT(11) NOT NULL COMMENT '作成者',
create_at DATETIME NOT NULL COMMENT '投稿時間'
)ENGINE=InnoDB ROW_FORMAT=DYNAMIC COLLATE=utf8mb4_general_ci COMMENT='練習アルバム';
